Правила редиректов для scripts/build-htaccess.mjs

Раздел «устаревшие адреса» (после RewriteRule ^shop/uncategorized/$). ${ORIGIN} — как в соседних правилах (https://miningpartsgroup.ru); в шаблонной строке build-htaccess.mjs обратный слэш удваивается (\\d), как в соседних правилах. QSD отбрасывает старые параметры ?_product_brand=…:

RewriteRule ^product-category/(chasti-privoda|detali-dvigatelya|detali-kuzova|detali-transmissii|elektrika|gidravlika|hodovaya-chast)(/page/\d+)?/?$ ${ORIGIN}/shop/$1/ [R=301,L,QSD]

RewriteRule ^product-category/(chasti-privoda|detali-dvigatelya|detali-kuzova|detali-transmissii|elektrika|gidravlika|hodovaya-chast)(/page/\\d+)?/?$ ${ORIGIN}/shop/$1/ [R=301,L,QSD]
RewriteRule ^product-category/ ${ORIGIN}/shop/ [R=301,L,QSD]
RewriteRule ^shop/page/\\d+/?$ ${ORIGIN}/shop/ [R=301,L,QSD]
RewriteRule ^brand/(berco|epiroc|sandvik|john-deere|kubota|prochee)/?$ ${ORIGIN}/shop/ [R=301,L,QSD]
